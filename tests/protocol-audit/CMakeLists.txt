cmake_minimum_required(VERSION 3.20)
project(PwarProtocolAudit LANGUAGES C CXX)
enable_testing()

set(PWAR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../..")
if(NOT EXISTS "${PWAR_SOURCE_DIR}/protocol/pwar_router.c")
    message(FATAL_ERROR "PWAR_SOURCE_DIR must identify the source under audit")
endif()
add_executable(PwarProtocolAudit protocol_audit.cpp "${PWAR_SOURCE_DIR}/protocol/pwar_router.c")
target_include_directories(PwarProtocolAudit PRIVATE "${PWAR_SOURCE_DIR}/protocol")
target_compile_features(PwarProtocolAudit PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(PwarProtocolAudit PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()
foreach(case partial duplicate inconsistent stale permutations streaming wrap loss reset dimensions deterministic sender_capacity)
    add_test(NAME PwarProtocol_${case} COMMAND PwarProtocolAudit ${case})
endforeach()
add_executable(PwarReceiveBufferAudit receive_buffer_audit.cpp "${PWAR_SOURCE_DIR}/protocol/pwar_rcv_buffer.c")
target_include_directories(PwarReceiveBufferAudit PRIVATE "${PWAR_SOURCE_DIR}/protocol")
target_compile_features(PwarReceiveBufferAudit PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(PwarReceiveBufferAudit PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()
foreach(case grow shrink tail invalid)
    add_test(NAME PwarReceive_${case} COMMAND PwarReceiveBufferAudit ${case})
endforeach()
set(PWAR_CLI_PATH "" CACHE FILEPATH "Optional real Linux CLI executable for argument checks")
if(NOT WIN32)
    find_package(Threads REQUIRED)
    add_executable(PwarAudioQueueTests audio_queue_tests.c
        "${PWAR_SOURCE_DIR}/protocol/pwar_audio_queue.c")
    set_property(TARGET PwarAudioQueueTests PROPERTY C_STANDARD 11)
    target_include_directories(PwarAudioQueueTests PRIVATE "${PWAR_SOURCE_DIR}/protocol")
    target_link_libraries(PwarAudioQueueTests Threads::Threads)
    add_test(NAME PwarAudioQueueTests COMMAND PwarAudioQueueTests)
    set_tests_properties(PwarAudioQueueTests PROPERTIES TIMEOUT 30)
endif()
if(PWAR_CLI_PATH)
    foreach(case help unknown port missing address buffer check)
        add_test(NAME PwarCli_${case} COMMAND ${CMAKE_COMMAND}
            -DCLI=${PWAR_CLI_PATH} -DCASE=${case} -P ${CMAKE_CURRENT_SOURCE_DIR}/cli_check.cmake)
    endforeach()
endif()
