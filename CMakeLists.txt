cmake_minimum_required(VERSION 3.20)
project(TimoxPwar LANGUAGES C CXX)
include(cmake/PwarVersion.cmake)
configure_file(cmake/pwar_version.h.in generated/pwar_version.h @ONLY)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
option(PWAR_BUILD_LINUX "Build Linux PipeWire backends" ${UNIX})
option(PWAR_BUILD_WINDOWS_CLIENT "Build standalone Windows ASIO client" ${WIN32})
option(PWAR_BUILD_LEGACY_DRIVER "Build legacy PWAR DLL, without registration" OFF)
add_subdirectory(tests)
if(PWAR_BUILD_LINUX)
    add_subdirectory(linux)
endif()
if(PWAR_BUILD_WINDOWS_CLIENT)
    if(NOT WIN32)
        message(FATAL_ERROR "PWAR Windows client requires Windows")
    endif()
    enable_language(RC)
    add_executable(PwarAsioClient client/asio_client.cpp client/pwar_version.rc protocol/pwar_router.c)
    target_include_directories(PwarAsioClient PRIVATE deps deps/asiosdk/common protocol "${CMAKE_BINARY_DIR}/generated")
    target_link_libraries(PwarAsioClient PRIVATE ws2_32 ole32 uuid Threads::Threads)
    if(MSVC)
        target_compile_options(PwarAsioClient PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
    endif()
    add_test(NAME PwarAsioClientHelp COMMAND PwarAsioClient --help)
    add_test(NAME PwarAsioClientVersion COMMAND PwarAsioClient --version)
    set_tests_properties(PwarAsioClientVersion PROPERTIES PASS_REGULAR_EXPRESSION "Timox PWAR ${PWAR_VERSION}")
    add_test(NAME PwarAsioClientInvalidArguments COMMAND PwarAsioClient --seconds nope)
    set_tests_properties(PwarAsioClientInvalidArguments PROPERTIES WILL_FAIL TRUE)
    add_custom_target(PwarBuildInfo ALL
        COMMAND ${CMAKE_COMMAND} "-DSOURCE_ROOT=${CMAKE_SOURCE_DIR}"
            "-DBINARY_FILE=$<TARGET_FILE:PwarAsioClient>" "-DOUTPUT_FILE=$<TARGET_FILE_DIR:PwarAsioClient>/pwar-build.json"
            "-DBINARY_PLATFORM=windows-x64" -P "${CMAKE_SOURCE_DIR}/cmake/WriteBuildInfo.cmake"
        DEPENDS PwarAsioClient VERBATIM)
endif()
if(PWAR_BUILD_LEGACY_DRIVER)
    if(NOT WIN32 OR NOT MSVC)
        message(FATAL_ERROR "Legacy PWAR driver requires Windows/MSVC")
    endif()
    add_subdirectory(windows/asio)
endif()
